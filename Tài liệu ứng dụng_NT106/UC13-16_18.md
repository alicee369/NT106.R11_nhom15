### Lấy danh sách kế hoạch (API Mới bổ sung cho UC13/UC1)
- URL Endpoint: /api/plan/list
- Method: POST
- Header: Authorization: Bearer <token>
--------------------------------------------------
// Xem danh sách kế hoạch & tiến độ
    //Client:
{
  "action": "PLAN_GET_LIST",
  "requestId": int,
  "data": {}
}
    //Server:
        //success
{
  "action": "PLAN_GET_LIST",
  "requestId": int,
  "success": true,
  "errorCode": null,
  "message": "Thành công.",
  "data": {
    "plans": [
      {
        "planId": int,
        "planName": str,
        "targetMinutes": int,
        "progressPercent": float,
        "studiedMinutes": int
      }
    ]
  }
}

### UC16: Xem thống kê trực quan
- URL Endpoint: /api/statistic/view
- Method: POST
--------------------------------------------------
// Xem thống kê (Đã bổ sung bộ lọc subjectTag theo yêu cầu)
    //Client:
{
  "action": "STATISTIC_GET_CHART",
  "requestId": int,
  "data": {
    "filterType": str, // "DAY" | "WEEK" | "MONTH"
    "timezone": str,
    "subjectTag": str // Truyền tên môn (VD: "Toán"). Để trống/null nếu lấy tổng.
  }
}
    //Server:
        //success
{
  "action": "STATISTIC_GET_CHART",
  "requestId": int,
  "success": true,
  "errorCode": null,
  "message": "Thành công.",
  "data": {
    "totalStudyMinutes": int,
    "chartData": [
      {
        "label": str, // VD: "Thứ 2" hoặc "2026-11-17"
        "subjectTag": str, 
        "studyMinutes": int
      }
    ]
  }
}

### UC18: Tính điểm và ước lượng GPA
- URL Endpoint: /api/score/estimate
- Method: POST
--------------------------------------------------
// Tính điểm
    //Client:
{
  "action": "SCORE_ESTIMATE_GPA",
  "requestId": int,
  "data": {
    "currentScores": [
      {
        "score": float,
        "weightPercent": float
      }
    ],
    "remainingWeightPercent": float,
    "targetGPA": float,
    "scale": float
  }
}
    //Server:
        //success 
{
  "action": "SCORE_ESTIMATE_GPA",
  "requestId": int,
  "success": true,
  "errorCode": null,
  "message": "Thành công.",
  "data": {
    "requiredScoreForRemainingWeight": float
  }
}

{
